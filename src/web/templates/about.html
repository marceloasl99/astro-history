{{ 'Ver Quickstart no README ↗' if lang == 'pt' else 'View Quickstart in README ↗' }} 
# 1. Clonar o repositório público do GitHub
git clone https://github.com/marceloasl99/astro-history.git
cd astro-history

# 2. Criar ambiente virtual e instalar dependências
python3 -m venv .venv && source .venv/bin/activate
pip install -r requirements.txt

# 3. Executar a suíte de 138 testes automatizados
pytest tests/

# 4. Iniciar a plataforma web local
uvicorn src.web.app:app --host 0.0.0.0 --port 8000
